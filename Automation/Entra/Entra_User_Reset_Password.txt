import json
import secrets
import string
import requests
from typing import Dict, Any
from urllib.parse import quote


GRAPH_BASE_URL = "https://graph.microsoft.com/v1.0"


def _get_access_token(
    tenant_id: str,
    client_id: str,
    client_secret: str
) -> str:
    """
    Get Microsoft Graph access token using
    OAuth 2.0 Client Credentials flow.
    """

    token_url = (
        f"https://login.microsoftonline.com/"
        f"{tenant_id}/oauth2/v2.0/token"
    )

    payload = {
        "client_id": client_id,
        "client_secret": client_secret,
        "scope": "https://graph.microsoft.com/.default",
        "grant_type": "client_credentials"
    }

    headers = {
        "Content-Type": "application/x-www-form-urlencoded"
    }

    try:

        response = requests.post(
            token_url,
            data=payload,
            headers=headers,
            timeout=60
        )

        if response.status_code >= 400:

            try:
                error_body = response.json()
            except ValueError:
                error_body = response.text

            raise Exception(
                "Microsoft Entra token request failed.\n"
                f"HTTP Status: {response.status_code}\n"
                f"Response: "
                f"{json.dumps(error_body, indent=2)}"
            )

        token_data = response.json()

        access_token = token_data.get(
            "access_token"
        )

        if not access_token:

            raise Exception(
                "Microsoft authentication response "
                "did not contain an access token."
            )

        return access_token

    except requests.exceptions.RequestException as e:

        error_body = ""

        if e.response is not None:

            try:
                error_body = e.response.json()
            except ValueError:
                error_body = e.response.text

        raise Exception(
            "Microsoft Entra authentication failed.\n"
            f"HTTP Status: "
            f"{e.response.status_code if e.response is not None else 'unknown'}\n"
            f"Error: {e}\n"
            f"Response: "
            f"{json.dumps(error_body, indent=2) if isinstance(error_body, dict) else error_body}"
        )


def _normalize_user(user) -> str:
    """
    Normalize the user value received from Exabeam AAM.

    Supports:

        "user@company.com"

    and:

        ["user@company.com"]

    Returns:

        "user@company.com"
    """

    if isinstance(user, list):

        if len(user) != 1:

            raise ValueError(
                f"Expected exactly one user, "
                f"but received: {user}"
            )

        user = user[0]

    if not isinstance(user, str):

        raise ValueError(
            f"Invalid user type: "
            f"{type(user).__name__}. "
            f"Expected string or single-element list."
        )

    user = user.strip()

    if not user:

        raise ValueError(
            "User cannot be empty."
        )

    return user


def _generate_temporary_password(
    length: int = 24
) -> str:
    """
    Generate a strong temporary password.

    The password contains:
        - uppercase letters
        - lowercase letters
        - numbers
        - special characters

    secrets is used instead of random for
    cryptographically secure generation.
    """

    if length < 16:

        raise ValueError(
            "Temporary password length must be "
            "at least 16 characters."
        )

    uppercase = string.ascii_uppercase
    lowercase = string.ascii_lowercase
    digits = string.digits
    special = "!@#$%^&*()-_=+"

    # Guarantee at least one character from each category.
    password_chars = [
        secrets.choice(uppercase),
        secrets.choice(lowercase),
        secrets.choice(digits),
        secrets.choice(special)
    ]

    all_characters = (
        uppercase +
        lowercase +
        digits +
        special
    )

    for _ in range(
        length - len(password_chars)
    ):
        password_chars.append(
            secrets.choice(all_characters)
        )

    # Securely shuffle the generated characters.
    secrets.SystemRandom().shuffle(
        password_chars
    )

    return "".join(password_chars)


def _get_user(
    access_token: str,
    user: str
) -> Dict[str, Any]:
    """
    Verify that the target Entra ID user exists.
    """

    user_encoded = quote(
        user,
        safe=""
    )

    graph_url = (
        f"{GRAPH_BASE_URL}/users/"
        f"{user_encoded}"
        "?$select=id,userPrincipalName,"
        "displayName,accountEnabled"
    )

    headers = {
        "Authorization": f"Bearer {access_token}",
        "Accept": "application/json"
    }

    try:

        response = requests.get(
            graph_url,
            headers=headers,
            timeout=60
        )

        if response.status_code >= 400:

            try:
                error_body = response.json()
            except ValueError:
                error_body = response.text

            raise Exception(
                "Microsoft Graph user lookup failed.\n"
                f"HTTP Status: {response.status_code}\n"
                f"URL: {graph_url}\n"
                f"Response: "
                f"{json.dumps(error_body, indent=2)}"
            )

        return response.json()

    except requests.exceptions.RequestException as e:

        error_body = ""

        if e.response is not None:

            try:
                error_body = e.response.json()
            except ValueError:
                error_body = e.response.text

        raise Exception(
            "Microsoft Graph user lookup request failed.\n"
            f"HTTP Status: "
            f"{e.response.status_code if e.response is not None else 'unknown'}\n"
            f"Error: {e}\n"
            f"Response: "
            f"{json.dumps(error_body, indent=2) if isinstance(error_body, dict) else error_body}"
        )


def _reset_password(
    access_token: str,
    user: str,
    temporary_password: str
) -> Dict[str, Any]:
    """
    Reset the user's password using Microsoft Graph.

    The user will be required to change the temporary
    password at the next sign-in.
    """

    user_encoded = quote(
        user,
        safe=""
    )

    graph_url = (
        f"{GRAPH_BASE_URL}/users/"
        f"{user_encoded}"
    )

    headers = {
        "Authorization": f"Bearer {access_token}",
        "Content-Type": "application/json"
    }

    payload = {
        "passwordProfile": {
            "password": temporary_password,
            "forceChangePasswordNextSignIn": True
        }
    }

    try:

        response = requests.patch(
            graph_url,
            headers=headers,
            json=payload,
            timeout=60
        )

        # Successful Microsoft Graph user update
        # normally returns HTTP 204.
        if 200 <= response.status_code < 300:

            return {
                "status": "success",
                "http_status": response.status_code,
                "user": user,
                "password_reset": True,
                "force_change_password_next_sign_in": True,
                "message": (
                    f"Microsoft Entra ID password for "
                    f"user '{user}' was successfully reset. "
                    f"The user must change the temporary "
                    f"password at the next sign-in."
                )
            }

        # Graph returned an error.
        try:
            error_body = response.json()
        except ValueError:
            error_body = response.text

        raise Exception(
            "Microsoft Graph returned an error while "
            "resetting the user's password.\n"
            f"HTTP Status: {response.status_code}\n"
            f"URL: {graph_url}\n"
            f"Response: "
            f"{json.dumps(error_body, indent=2)}"
        )

    except requests.exceptions.RequestException as e:

        error_body = ""

        if e.response is not None:

            try:
                error_body = e.response.json()
            except ValueError:
                error_body = e.response.text

        raise Exception(
            "Microsoft Graph password reset request failed.\n"
            f"HTTP Status: "
            f"{e.response.status_code if e.response is not None else 'unknown'}\n"
            f"Error: {e}\n"
            f"Response: "
            f"{json.dumps(error_body, indent=2) if isinstance(error_body, dict) else error_body}"
        )


def main(
    tenant_id: str,
    client_id: str,
    client_secret: str,
    user
) -> Dict[str, Any]:
    """
    Exabeam AAM entry point.

    Reset a Microsoft Entra ID user's password.

    Parameters:

        tenant_id
            Microsoft Entra Tenant ID.

        client_id
            Microsoft Entra App Registration
            Application / Client ID.

        client_secret
            Microsoft Entra App Registration
            Client Secret.

        user
            User Principal Name or Entra Object ID.

            Supported:

                "user@company.com"

            or:

                ["user@company.com"]

    Security behavior:

        A strong temporary password is generated
        automatically.

        The temporary password is NOT returned in
        the result to avoid exposing it in Exabeam
        logs.

        The user is forced to change the password
        at their next sign-in.
    """

    # ---------------------------------------------------------
    # Validate required parameters
    # ---------------------------------------------------------

    if not tenant_id:
        raise ValueError(
            "tenant_id is required."
        )

    if not client_id:
        raise ValueError(
            "client_id is required."
        )

    if not client_secret:
        raise ValueError(
            "client_secret is required."
        )

    if not user:
        raise ValueError(
            "user is required."
        )

    # ---------------------------------------------------------
    # Normalize Exabeam input
    # ---------------------------------------------------------

    normalized_user = _normalize_user(
        user
    )

    # ---------------------------------------------------------
    # Get Microsoft Graph token
    # ---------------------------------------------------------

    access_token = _get_access_token(
        tenant_id=tenant_id,
        client_id=client_id,
        client_secret=client_secret
    )

    # ---------------------------------------------------------
    # Verify target user exists
    # ---------------------------------------------------------

    user_information = _get_user(
        access_token=access_token,
        user=normalized_user
    )

    # ---------------------------------------------------------
    # Generate temporary password
    # ---------------------------------------------------------

    temporary_password = (
        _generate_temporary_password(
            length=24
        )
    )

    # ---------------------------------------------------------
    # Reset password
    # ---------------------------------------------------------

    result = _reset_password(
        access_token=access_token,
        user=normalized_user,
        temporary_password=temporary_password
    )

    # ---------------------------------------------------------
    # Add safe target-user information
    # ---------------------------------------------------------

    result["target_user"] = {
        "id": user_information.get("id"),
        "userPrincipalName": user_information.get(
            "userPrincipalName"
        ),
        "displayName": user_information.get(
            "displayName"
        ),
        "accountEnabled": user_information.get(
            "accountEnabled"
        )
    }

    return result
